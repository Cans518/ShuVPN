cmake_minimum_required(VERSION 3.16)
project(shuvpn_kde_proxy_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 QUIET COMPONENTS Core)
if(Qt6_FOUND)
    find_package(KF6KIO REQUIRED)
    set(SHUVPN_KIO_CORE KF6::KIOCore)
    set(SHUVPN_QT_CORE Qt6::Core)
else()
    find_package(Qt5 REQUIRED COMPONENTS Core)
    find_package(KF5KIO REQUIRED)
    set(SHUVPN_KIO_CORE KF5::KIOCore)
    set(SHUVPN_QT_CORE Qt5::Core)
endif()
add_executable(kde_proxy_consumer main.cpp)
target_link_libraries(kde_proxy_consumer PRIVATE ${SHUVPN_QT_CORE} ${SHUVPN_KIO_CORE})
